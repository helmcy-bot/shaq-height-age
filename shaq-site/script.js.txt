// Shaq growth data
const ages = [4, 6, 8, 10, 12, 13, 16, 21];
const heights = [118.1, 130.8, 144.1, 160.7, 177.8, 198.1, 200.7, 215.9];

// Estimate age using interpolation
function estimateAge(height) {
    if (height <= heights[0]) return ages[0];
    if (height >= heights[heights.length - 1]) return ages[ages.length - 1];

    for (let i = 0; i < heights.length - 1; i++) {
        const h0 = heights[i];
        const h1 = heights[i + 1];
        const a0 = ages[i];
        const a1 = ages[i + 1];

        if (height >= h0 && height <= h1) {
            const t = (height - h0) / (h1 - h0);
            return a0 + t * (a1 - a0);
        }
    }
}

// Convert decimal age → years & months
function ageToString(age) {
    const years = Math.floor(age);
    const months = Math.round((age - years) * 12);
    return `${years} years ${months} months`;
}

// Main calculate button
function calculate() {
    const height = parseFloat(document.getElementById("heightInput").value);
    const age = estimateAge(height);

    if (!age && age !== 0) {
        document.getElementById("result").innerHTML = "Invalid height.";
        return;
    }

    document.getElementById("result").innerHTML =
        `Estimated age: <strong>${ageToString(age)}</strong>`;
}

// 🔥 Animated Shaq scaling
function updateShaq() {
    const input = document.getElementById("heightInput");
    const value = parseFloat(input.value);
    const shaq = document.getElementById("shaqSilhouette");
    const label = document.getElementById("shaqHeightLabel");

    if (!value) return;

    // Scale silhouette: 215 cm = 100%, 120 cm = ~55%
    const scale = value / 215;

    shaq.style.transform = `scale(${scale})`;
    label.innerText = `${value} cm`;
}
